# "no mÃ¡ximo seis indicadores por linha" (seÃ§Ã£o 16.1) â€” itens extras quebram

MAX_METRICS_PER_ROW = 6

SMALL_SAMPLE_THRESHOLD = 15
MODERATE_SAMPLE_THRESHOLD = 40


def note(text: str, kind: str = "info") -> None:
    """Caixa de aviso com borda colorida por tipo (seÃ§Ã£o 5/13 do planejamento de UI/UX).

    Substitui os `<div class="note">...` inline que antes eram copiados e
    colados em `app.py`/`pages/1_Jogadores.py` â€” mesma aparÃªncia, uma
    definiÃ§Ã£o sÃ³ (`src/ui/theme.py`).
    """
    if kind not in _VALID_NOTE_KINDS:
        raise ValueError(f"kind invÃ¡lido: {kind!r}. Use um de {sorted(_VALID_NOTE_KINDS)}.")
    st.markdown(f'<div class="app-note app-note--{kind}">{text}</div>', unsafe_allow_html=True)


def db_last_updated(path: Path) -> datetime | None:
    """Data da Ãºltima escrita bem-sucedida no banco (proxy de "quando os dados
    foram atualizados"): a data de modificaÃ§Ã£o do arquivo SQLite. NÃ£o Ã© um
    timestamp exato de coleta guardado explicitamente â€” Ã© uma aproximaÃ§Ã£o
    simples, no mesmo espÃ­rito das outras aproximaÃ§Ãµes jÃ¡ documentadas no
    projeto (ex.: Rating oficial por pontos antes da coleta chegar).
    """
    path = Path(path)
    if not path.exists():
        return None
    return datetime.fromtimestamp(path.stat().st_mtime)


def source_badge(
    source_label: str,
    updated_at: datetime | pd.Timestamp | None,
    seasons: str | None = None,
) -> None:
    """Selo de fonte e atualizaÃ§Ã£o (seÃ§Ã£o 5): "Fonte: X Â· Atualizado em ... Â· Temporadas: Y".

    `updated_at=None` omite o trecho de data â€” a base demonstrativa nÃ£o tem
    uma data de coleta real para mostrar.
    """
    parts = [f"Fonte: {source_label}"]
    if updated_at is not None:
        parts.append(f"Atualizado em {format_datetime_br(updated_at)}")
    if seasons:
        parts.append(f"Temporadas: {seasons}")
    st.markdown(f'<div class="source-badge">{" Â· ".join(parts)}</div>', unsafe_allow_html=True)


def metric_row(items: list[dict]) -> None:
    """CartÃµes de indicadores padronizados (seÃ§Ã£o 16.1 do planejamento de UI/UX).

    Cada item: `{"label": str, "value": str, "delta": str opcional, "help": str opcional}`.
    Mais de `MAX_METRICS_PER_ROW` itens quebram em vÃ¡rias linhas de `st.columns`
    em vez de espremer tudo numa linha sÃ³.
    """
    for start in range(0, len(items), MAX_METRICS_PER_ROW):
        chunk = items[start : start + MAX_METRICS_PER_ROW]
        columns = st.columns(len(chunk))
        for column, item in zip(columns, chunk, strict=True):
            column.metric(item["label"], item["value"], item.get("delta"), help=item.get("help"))


def sample_size_caption(shown: int, total: int, unit: str = "partidas") -> None:
    """"Exibindo X de Y partidas Â· Amostra {pequena/moderada/grande}" (seÃ§Ã£o 7.1)."""
    if shown <= SMALL_SAMPLE_THRESHOLD:
        label = "pequena"
    elif shown <= MODERATE_SAMPLE_THRESHOLD:
        label = "moderada"
    else:
        label = "grande"
    st.caption(
        f"Exibindo {format_int_br(shown)} de {format_int_br(total)} {unit} Â· Amostra {label}"
    )


def clear_filters_button(keys: list[str], label: str = "Limpar filtros") -> None:
    """BotÃ£o que remove os widgets de filtro informados (por `key`) do
    `st.session_state` e recarrega a pÃ¡gina â€” cada widget volta ao seu valor
    padrÃ£o declarado (seÃ§Ã£o 7.2: "oferecer um botÃ£o Limpar filtros").
    """
    if st.button(label):
        for key in keys:
            st.session_state.pop(key, None)
        st.rerun()


def persistent_selectbox(
    label: str, options: list[str], query_key: str, default_index: int = 0, **kwargs
) -> str:
    """`st.selectbox` cujo valor escolhido fica salvo num parÃ¢metro da URL
    (seÃ§Ã£o "salvar preferÃªncias de filtros" do planejamento de UI/UX):
    recarregar a pÃ¡gina ou compartilhar o link mantÃ©m a mesma seleÃ§Ã£o, em vez
    de voltar sempre para a primeira opÃ§Ã£o.

    Se o valor salvo na URL nÃ£o existir mais em `options` (base mudou, opÃ§Ã£o
    saiu de cena), cai de volta em `default_index` silenciosamente â€” nunca
    quebra por causa de uma URL desatualizada.
    """
    if not options:
        raise ValueError("options nÃ£o pode ser vazio.")
    saved_value = st.query_params.get(query_key)
    index = options.index(saved_value) if saved_value in options else default_index
    selected = st.selectbox(label, options, index=index, **kwargs)
    st.query_params[query_key] = selected
    return selected

from __future__ import annotations

from datetime import datetime

import pandas as pd

# PadrÃ£o brasileiro (seÃ§Ã£o 14 do planejamento de UI/UX): separador de milhar
# com ponto, decimal com vÃ­rgula, data dd/mm/aaaa. O resto do app hoje usa o
# padrÃ£o americano nas tabelas/grÃ¡ficos jÃ¡ existentes (nÃ£o mexido aqui) â€”
# estes formatadores valem para os componentes novos de `src/ui`.


def format_int_br(value: int | float) -> str:
    """Milhar com ponto: 12345 -> "12.345"."""
    return f"{int(round(value)):,}".replace(",", ".")


def format_decimal_br(value: float, decimals: int = 1) -> str:
    """Decimal com vÃ­rgula: 23.8 -> "23,8"."""
    text = f"{value:,.{decimals}f}"
    return text.replace(",", "âŸ").replace(".", ",").replace("âŸ", ".")


def format_pct_br(value: float, decimals: int = 1) -> str:
    """Percentual com vÃ­rgula: 0.475 -> "47,5%" (`value` Ã© fraÃ§Ã£o, nÃ£o jÃ¡ Ã—100)."""
    return f"{format_decimal_br(value * 100, decimals)}%"


def format_date_br(value: datetime | pd.Timestamp | str) -> str:
    """Data no formato brasileiro: dd/mm/aaaa."""
    timestamp = pd.Timestamp(value)
    return timestamp.strftime("%d/%m/%Y")


def format_datetime_br(value: datetime | pd.Timestamp | str) -> str:
    """Data e hora no formato brasileiro: dd/mm/aaaa Ã s HH:MM."""
    timestamp = pd.Timestamp(value)
    return timestamp.strftime("%d/%m/%Y Ã s %H:%M")

def empty_state(title: str, description: str, actions: list[tuple[str, str]] | None = None) -> None:
    """Banco vazio (seÃ§Ã£o 13.1 do planejamento de UI/UX): tÃ­tulo curto + explicaÃ§Ã£o
    + sugestÃµes de aÃ§Ã£o em texto.

    Os botÃµes de aÃ§Ã£o em si continuam sendo responsabilidade de quem chama
    (cada pÃ¡gina sabe qual comando/link oferecer) â€” isto sÃ³ padroniza a
    moldura visual. Sempre termina com `st.stop()`: nada depois desta chamada

    st.info(f"**{title}**\n\n{description}")
    for label, detail in actions or []:
        st.caption(f"â€¢ **{label}** â€” {detail}")
    st.stop()


def error_state(message: str, detail: str | None = None) -> None:
    """Erro na fonte de dados (seÃ§Ã£o 13.3): o que falhou, com detalhe tÃ©cnico opcional."""
    st.error(message if detail is None else f"{message}\n\n`{detail}`")


def small_sample_warning(count: int, threshold: int = 15, unit: str = "jogos") -> None:
    """Amostra pequena (seÃ§Ã£o 13.4): "Apenas N jogos correspondem aos filtros...".

    SÃ³ emite o aviso quando `count <= threshold` â€” chamada segura mesmo
    quando a amostra jÃ¡ Ã© grande.
    """
    if count <= threshold:
        st.warning(
            f"Apenas {count} {unit} correspondem aos filtros. "
            "A projeÃ§Ã£o pode variar mais do que o normal."
        )


def stale_data_notice(last_updated: datetime | pd.Timestamp | None, days_threshold: int = 7) -> None:
    """Dados desatualizados (seÃ§Ã£o 13.5): avisa quando a Ãºltima atualizaÃ§Ã£o passou
    de `days_threshold` dias.

    Silencioso quando `last_updated` Ã© `None` (base sem data de coleta
    conhecida, ex.: demonstrativa).
    """
    if last_updated is None:
        return
    age_days = (pd.Timestamp.now() - pd.Timestamp(last_updated)).days
    if age_days > days_threshold:
        st.warning(
            f"Os dados foram atualizados hÃ¡ {age_days} dias "
            f"({format_datetime_br(last_updated)}). Alguns jogos recentes podem nÃ£o estar incluÃ­dos."
        )

def probability_chart(home_team: str, away_team: str, home_probability: float):
    chart_data = pd.DataFrame(
        {
            "Time": [home_team, away_team],
            "Probabilidade": [home_probability, 1 - home_probability],
        }
    )
    figure = px.bar(
        chart_data,
        x="Probabilidade",
        y="Time",
        orientation="h",
        text=chart_data["Probabilidade"].map(lambda value: f"{value:.1%}"),
        color="Time",
        color_discrete_sequence=["#F97316", "#38BDF8"],
    )
    figure.update_traces(textposition="inside", textfont_size=16)
    figure.update_layout(
        showlegend=False,
        height=250,
        xaxis_tickformat=".0%",
        xaxis_range=[0, 1],
        margin=dict(l=10, r=10, t=20, b=10),
        xaxis_title=None,
        yaxis_title=None,
    )
    return figure


def recent_form_chart(games: pd.DataFrame, home_team: str, away_team: str):
    frames = []
    for team in (home_team, away_team):
        history = team_game_history(games, team).tail(15).copy()
        history["Time"] = team
        history["MÃ©dia mÃ³vel de vitÃ³rias"] = history["win"].rolling(5, min_periods=1).mean()
        frames.append(history)
    data = pd.concat(frames, ignore_index=True)
    figure = px.line(
        data,
        x="game_date",
        y="MÃ©dia mÃ³vel de vitÃ³rias",
        color="Time",
        markers=True,
        color_discrete_sequence=["#F97316", "#38BDF8"],
    )
    figure.update_layout(
        height=330,
        yaxis_tickformat=".0%",
        yaxis_range=[0, 1],
        xaxis_title=None,
        yaxis_title="Forma (Ãºltimos 5)",
        legend_title=None,
        margin=dict(l=10, r=10, t=25, b=10),
    )
    return figure
